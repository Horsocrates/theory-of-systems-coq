```coq
Fixpoint wave_iter (s : Q) (N : nat) (u0 u1 : grid_fn) (n : nat) : grid_fn * grid_fn :=
  match n with
  | O => (u0, u1)
  | S k => (snd (wave_iter s N u0 u1 k),
            wave_step s N (fst (wave_iter s N u0 u1 k)) (snd (wave_iter s N u0 u1 k)))
  end.

(** Конечная скорость за n шагов: оба начальных уровня нулевые на [a,b] —
    оба уровня после n шагов нулевые на [a+n, b-n]. При всяком s и всяком N. *)
Theorem wave_finite_speed_n : forall s N (u0 u1 : grid_fn) a b,
  (forall k, (a <= k <= b)%nat -> u0 k == 0) ->
  (forall k, (a <= k <= b)%nat -> u1 k == 0) ->
  forall n j, (a + n <= j)%nat -> (j + n <= b)%nat ->
    fst (wave_iter s N u0 u1 n) j == 0 /\ snd (wave_iter s N u0 u1 n) j == 0.
Proof.
  intros s N u0 u1 a b H0 H1 n. induction n as [|n IH]; intros j Ha Hb.
  - cbn [wave_iter fst snd]. split; [apply H0 | apply H1]; lia.
  - cbn [wave_iter fst snd]. split.
    + apply (IH j); lia.
    + apply (wave_step_interior_zero s N _ _ (a + n)%nat (b - n)%nat).
      * intros k Hk. apply (IH k); lia.
      * intros k Hk. apply (IH k); lia.
      * lia.
      * lia.
Qed.

(** Область зависимости одного шага в общей форме. *)
Lemma wave_step_dep : forall s N (p1 c1 p2 c2 : grid_fn) j,
  c1 (ppred N j) == c2 (ppred N j) -> c1 j == c2 j -> c1 (S j) == c2 (S j) ->
  p1 j == p2 j ->
  wave_step s N p1 c1 j == wave_step s N p2 c2 j.
Proof.
  intros s N p1 c1 p2 c2 j Hm Hc HS Hp. unfold wave_step.
  rewrite Hm, Hc, HS, Hp. reflexivity.
Qed.

(** Конус зависимости за n шагов: начальные состояния, совпадающие на [a,b],
    дают состояния, совпадающие на [a+n, b-n]. *)
Theorem wave_dependence_cone_n : forall s N (u0 u1 v0 v1 : grid_fn) a b,
  (forall k, (a <= k <= b)%nat -> u0 k == v0 k) ->
  (forall k, (a <= k <= b)%nat -> u1 k == v1 k) ->
  forall n j, (a + n <= j)%nat -> (j + n <= b)%nat ->
    fst (wave_iter s N u0 u1 n) j == fst (wave_iter s N v0 v1 n) j /\
    snd (wave_iter s N u0 u1 n) j == snd (wave_iter s N v0 v1 n) j.
Proof.
  intros s N u0 u1 v0 v1 a b H0 H1 n. induction n as [|n IH]; intros j Ha Hb.
  - cbn [wave_iter fst snd]. split; [apply H0 | apply H1]; lia.
  - cbn [wave_iter fst snd]. split.
    + apply (IH j); lia.
    + destruct j as [|k]; [lia |].
      apply wave_step_dep.
      * cbn [ppred]. apply (IH k); lia.
      * apply (IH (S k)); lia.
      * apply (IH (S (S k))); lia.
      * apply (IH (S k)); lia.
Qed.
```

```coq
(** Энергия пары соседних уровней: сумма квадратов разности уровней и перекрёстный член. *)
Definition wave_energy (s : Q) (N : nat) (a b : grid_fn) : Q :=
  sum_Q_ns (fun j => (b j - a j) * (b j - a j)) N
  + s * sum_Q_ns (fun j => fd b j * fd a j) N.

(** Сохранение за один шаг. *)
Theorem wave_energy_step : forall s N (a b : grid_fn), (0 < N)%nat ->
  a N == a 0%nat -> b N == b 0%nat -> b (S N) == b 1%nat ->
  wave_energy s N b (wave_step s N a b) == wave_energy s N a b.

Definition gperiodic (N : nat) (u : grid_fn) : Prop := forall k, u (k + N)%nat == u k.

(** Сохранение за n шагов. *)
Theorem wave_energy_conserved : forall s N (u0 u1 : grid_fn) n, (0 < N)%nat ->
  gperiodic N u0 -> gperiodic N u1 ->
  wave_energy s N (fst (wave_iter s N u0 u1 n)) (snd (wave_iter s N u0 u1 n))
  == wave_energy s N u0 u1.

(** Оценка снизу и роль условия Куранта. *)
Theorem wave_energy_lower : forall s N (a b : grid_fn),
  0 <= s -> a N == a 0%nat -> b N == b 0%nat ->
  (1 - s) * sum_Q_ns (fun j => (b j - a j) * (b j - a j)) N <= wave_energy s N a b.

Corollary wave_energy_nonneg : forall s N (a b : grid_fn),
  0 <= s -> s <= 1 -> a N == a 0%nat -> b N == b 0%nat -> 0 <= wave_energy s N a b.

Theorem wave_cfl_increment_bound : forall s N (u0 u1 : grid_fn) n, (0 < N)%nat ->
  0 <= s -> gperiodic N u0 -> gperiodic N u1 ->
  (1 - s) * sum_Q_ns (fun j =>
       (snd (wave_iter s N u0 u1 n) j - fst (wave_iter s N u0 u1 n) j) *
       (snd (wave_iter s N u0 u1 n) j - fst (wave_iter s N u0 u1 n) j)) N
  <= wave_energy s N u0 u1.
```

   рациональном `s` (условие Куранта для сохранения не нужно), всяком `N > 0`, всяких `N`-периодических

   сохраняет периодичность (`wave_step_periodic`). Доказательство одного шага:

```coq
  set (c := wave_step s N a b).

  { unfold w, c, wave_step. destruct N as [|m]; [lia |]. cbn [ppred Nat.pred].
    rewrite Ha, Hb0, Hb1. reflexivity. }
  pose proof (wave_green N b w HN Hb0 Hb1 Hw) as HG.
  unfold wave_energy.
  assert (H1 : sum_Q_ns (fun j => (c j - b j) * (c j - b j)) N
             == sum_Q_ns (fun j => (b j - a j) * (b j - a j)) N
                + s * sum_Q_ns (fun j => w j * wlap N b j) N).
  { rewrite <- sum_ns_scale, <- sum_ns_add. apply sum_ns_ext. intros j _.
    unfold w, c, wave_step, wlap. ring. }
  assert (H2 : sum_Q_ns (fun j => fd c j * fd b j) N
             == sum_Q_ns (fun j => fd b j * fd a j) N
                + sum_Q_ns (fun j => fd b j * fd w j) N).
  { rewrite <- sum_ns_add. apply sum_ns_ext. intros j _. unfold w, fd. ring. }
  rewrite H1, H2, HG. ring.
Qed.
```

   Здесь `wlap N b j := b (S j) - 2 * b j + b (ppred N j)`, а `wave_green` — суммирование по частям в форме
   $\sum_j (Db)_j (Dw)_j = -\sum_j w_j (\Delta b)_j$ при `b N == b 0`, `b (S N) == b 1`, `w N == w 0`
   (`Check.v`, стр. 192–213).
2. **«Чувствительность к выбору энергии».** Энергия с потенциальной частью на одном уровне
   ($\sum (b-a)^2 + s\sum (Db)^2$) той же схемой не сохраняется: на периодическом импульсе, $N=5$, $s=1/2$, она
   равна 2 до шага и 1 после (`wave_energy_naive_not_conserved`); перекрёстная на тех же данных равна 1 и 1
   (`wave_energy_cross_example`).
3. **Условие Куранта.** При $0 \le s \le 1$ энергия неотрицательна, и сумма квадратов приращения по времени на
   всех шагах не превосходит $E_0/(1-s)$ (`wave_cfl_increment_bound`). При $s = 2$ на сетке из двух узлов
   энергия знакопеременной моды отрицательна (`wave_energy_negative_beyond_cfl`), а значения в узле растут:
   $1, -5, 29, -169$ (`wave_growth_beyond_cfl`); при $s=1$ та же мода остаётся в пределах единицы
   (`wave_bounded_at_cfl`).

Если эти теоремы будут внесены в формализацию (новым файлом; существующие не меняются), слова страницы о
недоказанности сохранения энергии (стр. 33–38, 43–44, 56–57, 142–162, 178–180, 187, 196–197, 207–216) перестанут
соответствовать положению дел; как их изменить — решение автора. Фактическая часть для такой правки:
сохраняется величина $E = \sum_j (u^{n+1}_j-u^n_j)^2 + s\sum_j (u^{n+1}_{j+1}-u^{n+1}_j)(u^{n}_{j+1}-u^{n}_j)$ —
на периодической сетке, при всяком $s$, для всяких начальных уровней и всякого числа шагов; при $0\le s\le 1$
она неотрицательна. Завершённая волна, непрерывный предел и дисперсия при этом остаются границей.

## Сомнительные места (расхождением не являются)

1. **Стр. 56, 178, сноска 6: запись условия Куранта «$c^2\le1$».** На странице $s = c^2(\Delta t/\Delta x)^2$ и
   «$s$ — роль-число-Куранта (квадрат)». Тогда условием на схему служит $s \le 1$ (см. раздел выше, п. 3);
   запись $c^2 \le 1$ совпадает с ним только при $\Delta t = \Delta x$ или если $c$ обозначает само число
   Куранта (в `WavePropagation.v` параметр схемы так и назван `c_sq`). То же написано в шапке
   `ProcessWaveEnergy.v`, и сноска 6 цитирует шапку верно. Теоремы с этим условием в файле нет, поэтому это
   замечание о записи, а не о формализации. Точнее было бы «$s\le1$».
2. **Граничные условия в определении шага.** Страница их не называет и говорит «конечная сетка» (стр. 65, 190,
   198). В `wave_step` левый сосед узла 0 — узел $N-1$ (`ppred`), правый сосед узла $j$ — узел $j+1$ без
   заворачивания; грид-функция есть `nat -> Q`, шаг определён при всех натуральных $j$, а $N$ входит только в
   левого соседа узла 0. Периодической схема становится для функций, продолженных с периодом $N$ (шаг такое
   продолжение сохраняет: `Check.v`, `wave_step_periodic`). Шесть лемм файла периодичности не берут и от неё
   не зависят; в лемме о тихой области $N$ роли не играет ($j \ge 1$). В примере главы правым соседом узла 4
   служит индекс 5 (значение `wimp 5` — нуль), левым соседом узла 0 — узел 4.
3. **Стр. 103–104: «она будет молчать в этом узле и дальше».** Лемма даёт нуль на одном следующем шаге; через
   два шага возмущение от дальних узлов может прийти. Формула под этими словами (стр. 106–109) точна.
4. **Стр. 130–132: узлы 3 и 4 примера.** В файле вычислены узлы 2, 1 и 0; сноска 5 называет ровно эти три
   леммы. Значения узлов 3 и 4 страница называет верно — $\tfrac12$ и $0$ (`Check.v`, `wave_node3`,
   `wave_node4`), но в формализации их нет.
5. **Стр. 98–101: область зависимости.** В файле она записана в форме нулей (`wave_step_local_zero`). Форма
   «состояния, совпадающие в трёх соседних узлах и в собственном прошлом, дают одно значение» следует из
   определения сразу (`Check.v`, `wave_step_dep`), но отдельной леммой в файле не стоит.
6. **Стр. 84–89: «Связь с Частью VIII здесь буквальна».** В формализации связи между схемой чехарды и
   осциллятором Главы 8.5 нет: там — явный метод Эйлера для пары (`ProcessODESystem.v`,
   `ProcessODELinearLaws.v`), здесь — центральная разность по времени; общих определений и лемм у двух файлов
   нет. Это чтение автора; как утверждение о формализации оно на странице не подано.
7. **Стр. 88–89: «Иерархия … прослежена в акустическом слое репозитория».** Слой есть (`src/acoustics/`, 10
   файлов). В `HierarchyTheorem.v` уровни заданы записями, вложение «волна содержит вибрацию»
   (`wave_has_vibration`, `:59`) есть поле записи, остальное — вычисления на цепочке из четырёх узлов;
   уровня «колебание» среди записей этого файла нет (он в `Oscillation.v`). Слово «прослежена» доказательства
   не заявляет.
8. **Сноска 1: «родственный ранний файл».** В `acoustics/WavePropagation.v` объявлено своё `wave_step` с тем же
   именем и порядком аргументов, но с другим правилом на краю (за краем сетки — нуль). Внутри сетки
   ($0 < v < N-1$) оба определения совпадают, на краю различны (`Check.v`, раздел 3:
   `wave_step_defs_agree_interior`, `wave_step_defs_differ_boundary` — $3/2$ против $2$). Инструмент
   `tom-status` относит имя `wave_step` страницы к акустическому файлу; в ведомости оно привязано к
   `ProcessWaveEnergy.v`, как и говорит страница.
9. **Сноска 2 при стр. 80–82.** Сноска называет определение и линейность; фраза, к которой она привязана,
   говорит о двух уровнях времени. Линейность доказана (`wave_step_linear`), но к этой фразе не относится.
10. **Решение Даламбера и характеристики.** Страница о них не говорит, в файле их нет. К сведению: при числе
    Куранта 1 схема переносит бегущие профили точно (`Check.v`, раздел 6: `wave_step_courant_one`,
    `wave_travelling_left`, `wave_travelling_right`), то есть сеточный аналог формулы Даламбера доводится без
    аксиом.
