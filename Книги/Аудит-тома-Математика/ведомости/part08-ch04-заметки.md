```coq
(** Дискретное неравенство Гронуолла: всякая последовательность, растущая не
    быстрее рекуррентного шага, не превосходит итерации этого шага. *)
Theorem gronwall_inequality_from : forall (alpha beta h : Q) (u : nat -> Q) (u0 : Q),
  0 <= 1 + alpha * h ->
  u 0%nat <= u0 ->
  (forall n, u (S n) <= (1 + alpha * h) * u n + beta * h) ->
  forall n, u n <= gronwall_iterate alpha beta h u0 n.
Proof.
  intros alpha beta h u u0 Hf H0 Hrec n. induction n as [|n IH].
  - cbn [gronwall_iterate]. exact H0.
  - cbn [gronwall_iterate]. unfold gronwall_step.
    eapply Qle_trans; [ apply Hrec | ].
    apply Qplus_le_compat; [ | apply Qle_refl ].
    apply qmul_le_l; assumption.
Qed.

Corollary gronwall_inequality : forall (alpha beta h : Q) (u : nat -> Q),
  0 <= 1 + alpha * h ->
  (forall n, u (S n) <= (1 + alpha * h) * u n + beta * h) ->
  forall n, u n <= gronwall_iterate alpha beta h (u 0%nat) n.

(** Однородный случай: слова стр. 55–56 одной теоремой. *)
Corollary gronwall_inequality_hom : forall (alpha h : Q) (u : nat -> Q),
  0 <= 1 + alpha * h ->
  (forall n, u (S n) <= (1 + alpha * h) * u n) ->
  forall n, u n <= u 0%nat * Qpow (1 + alpha * h) n.

(** Итерация шага с притоком имеет явный вид (равенство). *)
Lemma gronwall_iterate_closed : forall alpha beta h u0 n,
  ~ alpha == 0 ->
  gronwall_iterate alpha beta h u0 n == gronwall_bound alpha beta h u0 n.

(** Неравенство с притоком: явная оценка. *)
Corollary gronwall_inequality_forced : forall (alpha beta h : Q) (u : nat -> Q),
  0 < alpha -> 0 <= h ->
  (forall n, u (S n) <= (1 + alpha * h) * u n + beta * h) ->
  forall n, u n <= (u 0%nat + beta / alpha) * Qpow (1 + alpha * h) n - beta / alpha.

(** Две степенные функции совпадают. *)
Lemma qpow_is_Qpow : forall b n, qpow b n = Qpow b n.

(** Правая часть euler_diff_bound есть однородная итерация Гронуолла. *)
Lemma euler_bound_is_gronwall_iterate : forall L h d n,
  qpow (1 + L * h) n * d == gronwall_iterate L 0 h d n.

(** Устойчивость есть Гронуолл для расхождения (в посылках раздела Stability). *)
Theorem euler_divergence_gronwall : forall y0 z0 n,
  Qabs (euler f h y0 n - euler f h z0 n)
  <= gronwall_iterate L 0 h (Qabs (y0 - z0)) n.
Proof.
  intros y0 z0 n.
  apply (gronwall_inequality_from L 0 h
           (fun k => Qabs (euler f h y0 k - euler f h z0 k)) (Qabs (y0 - z0))).
  - exact one_plus_Lh_nonneg.
  - cbn [euler]. apply Qle_refl.
  - intro k. eapply Qle_trans; [ apply (euler_step_diff f L Hlip h Hh) | ].
    assert (E : (1 + L * h) * Qabs (euler f h y0 k - euler f h z0 k) + 0 * h
                == (1 + L * h) * Qabs (euler f h y0 k - euler f h z0 k)) by ring.
    rewrite E. apply Qle_refl.
Qed.
```

## Сомнительные места, расхождением не являющиеся

1. **Единственность и нулевое возмущение — два разных предмета (стр. 71–83).** `picard_unique` говорит о
   неподвижных точках сеточного оператора Пикара; оценка расхождения — о ломаных Эйлера. В `src` они не
   связаны: `ProcessODEStability.v` не подключает `ProcessPicardOperator.v`, следствия «равные начала дают
   равные траектории» отдельной теоремой нет. Страница называет связь «наглядным прочтением», а не теоремой,
   поэтому записано `ok`. Связь доказывается (`Check.v`, раздел 3, Closed under the global context):
   ломаная Эйлера неподвижна под оператором Пикара (`euler_is_picard_fixed`, без всяких посылок); всякая
   неподвижная точка совпадает на сетке с ломаной Эйлера (`picard_fixed_is_euler`, посылка — липшицевость);
   две неподвижные точки с началами $y_0$, $z_0$ расходятся не более чем в $(1+Lh)^k$ раз от
   $\lvert y_0-z_0\rvert$ (`picard_fixed_stability`); при $y_0=z_0$ они совпадают
   (`picard_unique_as_zero_perturbation`). Последние две теоремы и есть «единственность — устойчивость при
   нулевом возмущении» в виде утверждения. При этом единственность на сетке получается без условия сжатия
   $LT<1$ и без условия $L>0$ (`picard_unique_any_interval`): каждое значение неподвижной точки вычисляется
   из предыдущих. Слова курсива «процесс сжатия не оставляет места второй траектории» (стр. 82–83) для сетки
   точнее читать через это: траектория определена началом шаг за шагом.
2. **«На конечном интервале множитель конечен» (стр. 105–109, 114).** В утверждениях `ProcessODEStability.v`
   отрезка нет: число шагов $n$ и шаг $h$ независимы, множитель `qpow (1 + L * h) n` при каждом $n$ —
   рациональное число. Оценки, не зависящей от числа шагов на отрезке длины $t$, в `src` нет. Она доказана в
   пробном файле: при $LT<1$ — `euler_stability_uniform` (расхождение не больше $\lvert\Delta_0\rvert/(1-LT)$),
   при всяком $t$ — `euler_divergence_uniform`. Форма «для всякого $\varepsilon$ найдётся $\delta$» при данном
   $n$ — `euler_continuous_dependence`. Страница теоремой эти слова не называет.
3. **«Расхождение решений» (стр. 12–14, 50–51, 178).** Доказанная оценка относится к двум ломаным Эйлера с
   одним шагом. Глава в местах, где говорит о доказанном, называет предмет точно («двух Эйлеровых
   траекторий», стр. 27, 87); слова о решениях стоят в постановке вопроса и в названии системы.
4. **«Убывает» (стр. 139).** Доказано $u_n\le u_0$; невозрастание от шага к шагу доказано в пробном файле
   (`gronwall_iterate_decay_monotone`). «Два его режима ($\alpha$ положительно или отрицательно)» (стр. 144):
   затухание в файле доказано при $0<\lvert\alpha\rvert h<1$ (`gronwall_decay_bound`); при $\alpha h<-1$
   множитель отрицателен, и итерация меняет знак.
5. **Сноска 2.** Посылка $LT<1$ в сноске не названа («если две грид-функции неподвижны … они совпадают»); в
   тексте, к которому сноска относится, сказано «у сжимающего оператора». «Во всех узлах» — в узлах $k\le N$.
6. **Сноска 6, «это правая часть `euler_diff_bound`».** Правая часть `euler_diff_bound` —
   `qpow (1 + L * h) n * Qabs (y0 - z0)`, в `gronwall_beta_zero` стоит `u0 * Qpow (1 + alpha * h) n`: другой
   порядок множителей и другая степенная функция. Равенство верно (`Check.v`, `qpow_is_Qpow`,
   `euler_bound_is_gronwall_iterate`), но в `src` не записано — расхождение 1.
7. **Сноска 5, «соответствующий `RealProcess`».** `RealProcess` — тип всех последовательностей `nat -> Q`
   (`process/ProcessCore.v:34`); условие Коши в него не входит, и `gronwall_process` при $\alpha h>0$,
   $u_0>0$ не ограничен. Сноска сходимости и не утверждает. (То же отмечено в Части VI.)
8. **Роли в шапке и в тексте.** В шапке `ProcessODEStability.v` $\lvert\Delta_n\rvert$ — «роль-расхождение»,
   $y_0-z_0$ — «роль-возмущение»; в тексте главы — «роль-чувствительность» и «роль-причина». Сноска 9
   повторяет шапку.
9. **Стр. 158, «разбирался в Главах 8.2 и 8.5».** Глава 8.5 идёт после Главы 8.4; адреса верны (8.2 — раздел
   о $(1+1/N)^N$, 8.5 — `exp1_process`, `expt_process`).
10. **«Батч C» (сноска 8).** Слово из шапок файлов; на страницах тома не пояснено.
11. **Посылки раздела `Stability` выполнимы, оценка точна.** Для $f(t,y)=Ly$ при $L\ge0$, $h\ge0$ условие
    Липшица выполнено (`linear_is_lipschitz`), а расхождение равно $(1+Lh)^n\lvert y_0-z_0\rvert$
    (`euler_diff_bound_sharp`). Теорем под невыполнимой посылкой в четырёх файлах главы не найдено.

## Вне главы

- **К Главе 8.3.** (i) Единственность неподвижной точки сеточного оператора Пикара не требует условия сжатия:
  `picard_unique` (`ProcessPicardOperator.v:183`) верна без посылок `0 < L` и `L * Tlen < 1`
  (`Check.v`, `picard_unique_any_interval`); неподвижная точка есть ломаная Эйлера
  (`picard_fixed_is_euler`, `euler_is_picard_fixed`), итерации Пикара на сетке неподвижны после $N$ шагов
  (`picard_iter_exact`). (ii) Леммы, названные в сноске 3 Главы 8.3 как наследующие `classic`
  (`iterate_is_cauchy`, `Qpow_vanish`), по компилятору — Closed under the global context; `Qpow_vanish` лежит в
  `SeriesConvergence.v:164`, а не в `FixedPoint.v`. Шапка `FixedPoint.v` (стр. 12): «AXIOMS: classic (inherited
  from MonotoneConvergence via Qpow_vanish)» — устарела.
- **`ProcessGronwall.v`.** Второй заголовок-комментарий (стр. 24): «STATUS: 22 Qed» — в файле 18 (первая
  шапка называет 18). Комментарий стр. 13–14 формулирует неравенство («if u(n+1) ≤ … then u(n) ≤ …»), теорема
  `discrete_gronwall` (`:183`) говорит об итерации, заданной равенством, и является равенством. У
  `gronwall_decay` (`:138`) четыре посылки в доказательстве не используются. `ode_stability_bound` (`:261`,
  комментарий «Two solutions with different data: difference bounded by Gronwall») и `ode_stability_vanish`
  (`:270`) — утверждения о числе `diff0 * Qpow r n`; решений в них нет. `gronwall_is_process_construction`
  (`:296`) — существование процесса, равного `gronwall_bound`; свидетель — само определение. `Qpow_nonneg`,
  `Qpow_le_1`, `Qpow_compat` повторяют леммы `SeriesConvergence.v` (`Qpow_nonneg`, `Qpow_bound_1`, `Qpow_wd`).
- **Два объявления `discrete_gronwall` с разными утверждениями:** `process/ProcessGronwall.v:183` и
  `navier_stokes/GronwallAnalysis.v:148`. У второго комментарий («y(n) <= y(0)*g^n») расходится с
  утверждением (посылка `y 0 <= 1`, оценка `iterated_growth g n`).
- **`ProcessODEStability.v`.** Комментарий стр. 106–109 называет следствие, которого в файле нет. Шапка
  (стр. 26–27): «The COMPLETED e^{Lt} … (classic-based as a completed transcendental)» — теоремы с такой ценой
  нет. Файл не подключается ни одним файлом `src`. `stability_mult_8` (`:115`) — вычисление $2^3=8$.
- **`ProcessExpProcess.v`.** Шапка (стр. 21–25) относит к границе полную монотонность, оценку «< 3 for all N»
  и свойство Коши («the last classic-based»). В пробном файле все три доказаны без аксиом: `expt_monotone`
  (при $s=1$ — монотонность $e$-процесса), `exp1_below_3_all` (`forall N, exp1_process N < 3`),
  `expt_process_is_Cauchy`. Это касается Глав 8.2 и 8.5.
- **`ProcessODE.v`** (глава его не называет): `ode_continuous_dependence` (`:258`) и
  `ode_perturbation_vanish` (`:266`) — те же утверждения о числе `diff0 * Qpow r n`; `ode_error_gronwall`
  (`:219`) повторяет `gronwall_beta_zero` с двумя неиспользуемыми посылками; первый член `picard_lindelof`
  (`:180`) — `exists proc : RealProcess, is_Cauchy proc` — выполняется для постоянного процесса.
